# Respaldo diario con rotación de 14 días
$destino = 'E:\Respaldos'
foreach ($origen in $cfg.Origenes) {
    robocopy $origen.Ruta $tmp /E /R:2 /W:5
    Compress-Archive "$tmp\*" "$destino\$fecha.zip"
}
Get-ChildItem $destino -Filter *.zip |
    Where-Object LastWriteTime -lt (Get-Date).AddDays(-14) |
    Remove-Item
